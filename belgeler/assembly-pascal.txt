-----------------------------------------------------------------------------------
  1. assembly dilinden pascal diline parametre ile çağrı örneği:
-----------------------------------------------------------------------------------

begin
	asm
	mov eax,1   // soldan 1. parametre
	mov edx,2   // soldan 2. parametre
	mov ecx,3   // soldan 3. parametre
	call ParametreKontrol
end;

function ParametreKontrol(ADeger1, ADeger2, ADeger3: TKimlik): TISayi4;
begin

  SISTEM_MESAJ(RENK_SIYAH, '1. Parametre = %d', [ADeger1]);
  SISTEM_MESAJ(RENK_SIYAH, '2. Parametre = %d', [ADeger2]);
  SISTEM_MESAJ(RENK_SIYAH, '3. Parametre = %d', [ADeger3]);
end;

-----------------------------------------------------------------------------------
  3. object nesnelerine erişmek
-----------------------------------------------------------------------------------
type
  PGorev = ^TGorev	
  TGorev = record
    Deger1: TSayi4;   
  end;

type
  TGorevler = object
  private
    FGorevListesi: array[0..USTSINIR_GOREVSAYISI - 1] of PGorev;
    function GorevAl(ASiraNo: TISayi4): PGorev;
    procedure GorevYaz(ASiraNo: TISayi4; AGorev: PGorev);
  public
    constructor Create;
    property Gorev[ASiraNo: TISayi4]: PGorev read GorevAl write GorevYaz;
  end;
  
var
  GGorevler: TGorevler;
  
  // Değer1'in değerine erişmek
  mov   eax,1  // 1. dizi değeri
  shl   eax,2
  mov   esi,GGorevler.Gorev[eax]
  mov   eax,[esi + TGorev.Deger1]

-----------------------------------------------------------------------------------
  4. class nesnelerine erişmek
-----------------------------------------------------------------------------------
type
  PGorev = ^TGorev	
  TGorev = class
    Deger2: TSayi4;   
  end;

type
  TGorevler = class
  private
	Deger1: TSayi4;     
    FGorevListesi: array[0..USTSINIR_GOREVSAYISI - 1] of PGorev;
    function GorevAl(ASiraNo: TISayi4): PGorev;
    procedure GorevYaz(ASiraNo: TISayi4; AGorev: PGorev);
  public
    constructor Create;
    property Gorev[ASiraNo: TISayi4]: PGorev read GorevAl write GorevYaz;
	procedure Calistir;
  end;
  
var
  GGorevler: TGorevler;
  
  // Değer1'in değerine erişmek
  mov   esi,[GGorevler]
  mov   eax,[esi + TGorevler.Deger1]

  // Değer2'nin değerine erişmek
  mov   esi,[GGorevler]
  mov   eax,1  // 1. dizi değeri
  imul  eax, SizeOf(TGorev)
  mov   esi,[esi + TGorevler.FGorevListesi + eax]
  mov   eax,[esi + TGorev.Deger2]

  // işlev çalıştırmak
  mov   eax,GZamanlayicilar
  call  TGorevler.Calistir
